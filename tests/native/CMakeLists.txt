cmake_minimum_required(VERSION 3.22)
project(LegacyNativeTests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
if(APPLE AND NOT DEFINED CMAKE_CXX23_STANDARD_COMPILE_OPTION)
 set(CMAKE_CXX23_STANDARD_COMPILE_OPTION "-std=c++2b")
 set(CMAKE_CXX23_EXTENSION_COMPILE_OPTION "-std=gnu++2b")
endif()
include(FetchContent)
FetchContent_Declare(teslable GIT_REPOSITORY https://github.com/yoziru/tesla-ble.git GIT_TAG 42b899b7816cc3700b7361e65b526fdc04d19a51)
FetchContent_MakeAvailable(teslable)
get_filename_component(APP_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_executable(legacy-telemetry-harness legacy_telemetry_harness.cpp)
target_include_directories(legacy-telemetry-harness PRIVATE "${APP_ROOT}/app/src/main/cpp")
target_link_libraries(legacy-telemetry-harness PRIVATE TeslaBLE)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
enable_testing()
add_test(NAME legacy-telemetry-json COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/check_legacy_telemetry.py" $<TARGET_FILE:legacy-telemetry-harness>)

add_executable(legacy-extra-harness legacy_extra_harness.cpp)
target_include_directories(legacy-extra-harness PRIVATE "${APP_ROOT}/app/src/main/cpp")
target_link_libraries(legacy-extra-harness PRIVATE TeslaBLE)
add_test(NAME legacy-extra-commands COMMAND legacy-extra-harness)
